"""
Incremental Data Loader for Flight Price Pipeline
Handles change detection and incremental loading to PostgreSQL
"""

import pandas as pd
import hashlib
from datetime import datetime
from airflow.providers.postgres.hooks.postgres import PostgresHook
from airflow.providers.mysql.hooks.mysql import MySqlHook

class IncrementalDataLoader:
    """
    Handles incremental data loading with change detection
    """
    
    def __init__(self, postgres_conn_id='postgres_analytics', mysql_conn_id='mysql_staging'):
        self.postgres_hook = PostgresHook(postgres_conn_id=postgres_conn_id)
        self.mysql_hook = MySqlHook(mysql_conn_id=mysql_conn_id)
        self.load_stats = {
            'new': 0,
            'updated': 0,
            'deleted': 0,
            'unchanged': 0
        }
        
    def calculate_record_hash(self, row):
        """
        Calculate MD5 hash of record for change detection
        Uses key fields that define record uniqueness
        """
        key_fields = [
            str(row.get('airline', '')),
            str(row.get('source', '')),
            str(row.get('destination', '')),
            str(row.get('date', '')),
            str(row.get('departure_time', '')),
            str(row.get('base_fare', '')),
            str(row.get('total_fare', ''))
        ]
        
        record_string = '|'.join(key_fields)
        return hashlib.md5(record_string.encode()).hexdigest()
    
    def load_new_data_from_mysql(self):
        """Load validated data from MySQL staging"""
        query = "SELECT * FROM validated_flight_data WHERE is_valid = 1"
        df = self.mysql_hook.get_pandas_df(query)
        print(f"📥 Loaded {len(df):,} records from MySQL staging")
        return df
    
    def load_existing_data_from_postgres(self):
        """Load current active data from PostgreSQL Bronze layer"""
        query = """
        SELECT 
            airline, source, destination, date, departure_time,
            base_fare, total_fare, record_hash
        FROM bronze.validated_flights
        WHERE is_active = TRUE
        """
        
        try:
            df = self.postgres_hook.get_pandas_df(query)
            print(f"📊 Loaded {len(df):,} existing records from PostgreSQL")
            return df
        except Exception as e:
            print(f"⚠️ No existing data or error: {e}")
            return pd.DataFrame()
    
    def detect_changes(self, new_df):
        """
        Detect new, updated, and deleted records
        
        Returns:
            dict: {
                'new_records': DataFrame,
                'deleted_records': DataFrame,
                'unchanged_records': DataFrame,
                'change_percentage': float
            }
        """
        print("🔍 Detecting changes in dataset...")
        start_time = datetime.now()
        
        # Calculate hashes for new data
        new_df['record_hash'] = new_df.apply(self.calculate_record_hash, axis=1)
        
        # Get existing data
        existing_df = self.load_existing_data_from_postgres()
        
        if existing_df.empty:
            print("📊 No existing data. All records are new.")
            return {
                'new_records': new_df,
                'deleted_records': pd.DataFrame(),
                'unchanged_records': pd.DataFrame(),
                'change_percentage': 100.0
            }
        
        # Detect changes using hash comparison
        existing_hashes = set(existing_df['record_hash'])
        new_hashes = set(new_df['record_hash'])
        
        # New records: in new data but not in existing
        new_record_hashes = new_hashes - existing_hashes
        new_records = new_df[new_df['record_hash'].isin(new_record_hashes)]
        
        # Deleted records: in existing but not in new data
        deleted_record_hashes = existing_hashes - new_hashes
        deleted_records = existing_df[existing_df['record_hash'].isin(deleted_record_hashes)]
        
        # Unchanged records
        unchanged_hashes = existing_hashes & new_hashes
        unchanged_records = new_df[new_df['record_hash'].isin(unchanged_hashes)]
        
        # Calculate change percentage
        total_changes = len(new_records) + len(deleted_records)
        change_percentage = (total_changes / len(new_df)) * 100 if len(new_df) > 0 else 0
        
        # Update stats
        self.load_stats['new'] = len(new_records)
        self.load_stats['deleted'] = len(deleted_records)
        self.load_stats['unchanged'] = len(unchanged_records)
        
        detection_time = (datetime.now() - start_time).total_seconds()
        
        print(f"📊 Change Detection Summary:")
        print(f"   New records: {len(new_records):,}")
        print(f"   Deleted records: {len(deleted_records):,}")
        print(f"   Unchanged records: {len(unchanged_records):,}")
        print(f"   Change percentage: {change_percentage:.2f}%")
        print(f"   Detection time: {detection_time:.2f}s")
        
        return {
            'new_records': new_records,
            'deleted_records': deleted_records,
            'unchanged_records': unchanged_records,
            'change_percentage': change_percentage
        }
    
    def apply_incremental_load(self, changes, load_type='INCREMENTAL'):
        """
        Apply incremental changes to Bronze layer
        
        Args:
            changes (dict): Output from detect_changes()
            load_type (str): 'INCREMENTAL' or 'FULL'
        """
        print(f"🔄 Applying {load_type} load...")
        start_time = datetime.now()
        
        conn = self.postgres_hook.get_conn()
        cursor = conn.cursor()
        
        try:
            # Start transaction
            cursor.execute("BEGIN")
            
            # Insert new records
            if not changes['new_records'].empty:
                print(f"➕ Inserting {len(changes['new_records']):,} new records...")
                
                # Add ingestion timestamp
                changes['new_records']['ingestion_timestamp'] = datetime.now()
                changes['new_records']['is_active'] = True
                
                # Insert to PostgreSQL
                changes['new_records'].to_sql(
                    'validated_flights',
                    self.postgres_hook.get_sqlalchemy_engine(),
                    schema='bronze',
                    if_exists='append',
                    index=False,
                    method='multi',
                    chunksize=1000
                )
            
            # Soft delete removed records
            if not changes['deleted_records'].empty:
                print(f"🗑️ Marking {len(changes['deleted_records']):,} records as inactive...")
                deleted_hashes = changes['deleted_records']['record_hash'].tolist()
                
                # Build parameterized query
                placeholders = ','.join(['%s'] * len(deleted_hashes))
                update_query = f"""
                    UPDATE bronze.validated_flights
                    SET is_active = FALSE
                    WHERE record_hash IN ({placeholders})
                """
                
                cursor.execute(update_query, deleted_hashes)
            
            # Log metadata
            execution_time = int((datetime.now() - start_time).total_seconds())
            
            cursor.execute("""
                INSERT INTO bronze.data_load_metadata 
                (records_inserted, records_updated, records_deleted, records_unchanged,
                 load_type, change_percentage, execution_time_seconds)
                VALUES (%s, %s, %s, %s, %s, %s, %s)
            """, (
                self.load_stats['new'],
                self.load_stats['updated'],
                self.load_stats['deleted'],
                self.load_stats['unchanged'],
                load_type,
                changes['change_percentage'],
                execution_time
            ))
            
            # Commit transaction
            conn.commit()
            print(f"✅ {load_type} load completed successfully in {execution_time}s")
            
        except Exception as e:
            conn.rollback()
            print(f"❌ {load_type} load failed: {e}")
            raise
        finally:
            cursor.close()
            conn.close()
    
    def apply_full_load(self, df):
        """
        Perform full truncate and reload
        
        Args:
            df (pd.DataFrame): Complete dataset to load
        """
        print("🔄 Applying FULL load (truncate and reload)...")
        start_time = datetime.now()
        
        conn = self.postgres_hook.get_conn()
        cursor = conn.cursor()
        
        try:
            # Start transaction
            cursor.execute("BEGIN")
            
            # Truncate table
            print("🗑️ Truncating bronze.validated_flights...")
            cursor.execute("TRUNCATE TABLE bronze.validated_flights")
            
            # Add tracking columns
            df['record_hash'] = df.apply(self.calculate_record_hash, axis=1)
            df['ingestion_timestamp'] = datetime.now()
            df['is_active'] = True
            
            # Insert all data
            print(f"➕ Inserting {len(df):,} records...")
            df.to_sql(
                'validated_flights',
                self.postgres_hook.get_sqlalchemy_engine(),
                schema='bronze',
                if_exists='append',
                index=False,
                method='multi',
                chunksize=1000
            )
            
            # Log metadata
            execution_time = int((datetime.now() - start_time).total_seconds())
            
            cursor.execute("""
                INSERT INTO bronze.data_load_metadata 
                (records_inserted, load_type, change_percentage, execution_time_seconds)
                VALUES (%s, %s, %s, %s)
            """, (len(df), 'FULL', 100.0, execution_time))
            
            # Commit transaction
            conn.commit()
            print(f"✅ FULL load completed successfully in {execution_time}s")
            
        except Exception as e:
            conn.rollback()
            print(f"❌ FULL load failed: {e}")
            raise
        finally:
            cursor.close()
            conn.close()
    
    def get_load_statistics(self, days=30):
        """Get load statistics for monitoring"""
        query = f"""
        SELECT 
            load_timestamp,
            load_type,
            records_inserted,
            records_deleted,
            records_unchanged,
            change_percentage,
            execution_time_seconds
        FROM bronze.data_load_metadata
        WHERE load_timestamp >= NOW() - INTERVAL '{days} days'
        ORDER BY load_timestamp DESC
        LIMIT 10
        """
        
        return self.postgres_hook.get_pandas_df(query)

def transfer_to_postgres_incremental(**context):
    """
    Transfer validated data from MySQL to PostgreSQL using incremental loading
    Decides between FULL and INCREMENTAL load based on change percentage
    """
    from utils.incremental_loader import IncrementalDataLoader
    
    print("🚀 Starting incremental data transfer...")
    
    # Initialize loader
    loader = IncrementalDataLoader(
        postgres_conn_id='postgres_analytics',
        mysql_conn_id='mysql_staging'
    )
    
    # Load new data from MySQL
    df_new = loader.load_new_data_from_mysql()
    
    if df_new.empty:
        print("⚠️ No data to transfer")
        return
    
    # Detect changes
    changes = loader.detect_changes(df_new)
    
    # Decision: Full or Incremental load
    change_percentage = changes['change_percentage']
    FULL_LOAD_THRESHOLD = 50.0  # If >50% changed, do full reload
    
    if change_percentage > FULL_LOAD_THRESHOLD:
        # Full reload
        print(f"⚠️ {change_percentage:.1f}% of data changed (>{FULL_LOAD_THRESHOLD}%)")
        print("   Performing FULL load (truncate and reload)...")
        loader.apply_full_load(df_new)
        load_type = 'FULL'
    else:
        # Incremental load
        print(f"✅ {change_percentage:.1f}% of data changed (<={FULL_LOAD_THRESHOLD}%)")
        print("   Performing INCREMENTAL load...")
        loader.apply_incremental_load(changes, load_type='INCREMENTAL')
        load_type = 'INCREMENTAL'
    
    # Log statistics
    stats = loader.get_load_statistics(days=7)
    print("\n📊 Recent Load Statistics:")
    print(stats.to_string(index=False))
    
    # Push to XCom for downstream tasks
    context['ti'].xcom_push(key='load_type', value=load_type)
    context['ti'].xcom_push(key='records_inserted', value=loader.load_stats['new'])
    context['ti'].xcom_push(key='change_percentage', value=change_percentage)
    
    print(f"\n✅ Data transfer completed using {load_type} load")
